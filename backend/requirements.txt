# --- Vision pipeline -------------------------------------------------------
Pillow==11.0.0

# Object detection (Sprint 3). ultralytics ships the YOLOv8 family; the
# yolov8n.pt weights (~6 MB) auto-download from the ultralytics CDN on the
# first inference call and are cached under ~/.config/Ultralytics (or
# %APPDATA%\Ultralytics on Windows). ultralytics pulls in torch + torchvision
# + opencv automatically; recent 8.3.x releases have Python 3.13 wheels for
# all of them. Open lower bound so pip resolves a 3.13-compatible torch.
ultralytics>=8.3.0

# --- Voice pipeline --------------------------------------------------------
gTTS==2.5.4
# faster-whisper instead of openai-whisper:
#   - openai-whisper's setup.py uses `locals()` after exec(), which broke on
#     Python 3.13 (PEP 667). Every version is affected.
#   - faster-whisper ships pre-built wheels, uses the same Whisper models
#     under the hood (downloaded from HuggingFace on first run), runs ~4x
#     faster on CPU via CTranslate2, and doesn't pull in torch.
# Open lower bound so pip picks the latest, which pulls in a recent PyAV
# (>=13) that has Python 3.13 Windows wheels. The earlier 1.0.3 pin pulled
# a too-old av that wouldn't wheel-install on 3.13.
faster-whisper>=1.1.0
# Belt-and-braces: explicitly require an av that has 3.13 wheels.
# (We also use PyAV directly to decode PTT audio blobs in speech_service.py,
# avoiding the need for a system-installed ffmpeg.exe on PATH.)
av>=13.0.0

rapidfuzz==3.10.1
